3. Pilih Suara Bahasa Indonesia Secara Manual

Update fungsi speakNumber seperti ini agar suara Indonesia digunakan jika tersedia:

javascript
function speakNumber(number) {
  if (!('speechSynthesis' in window)) {
    console.warn('Speech Synthesis API tidak didukung di browser ini.');
    return;
  }

  const letter = number.charAt(0);
  const digits = number.slice(1).split('');

  const letterMap = {
    'A': 'A', 'B': 'B', 'C': 'C', 'D': 'D', 'E': 'E',
    'F': 'F', 'G': 'G', 'H': 'H', 'I': 'I', 'J': 'J',
    'K': 'K', 'L': 'L', 'M': 'M', 'N': 'N', 'O': 'O',
    'P': 'P', 'Q': 'Q', 'R': 'R', 'S': 'S', 'T': 'T',
    'U': 'U', 'V': 'V', 'W': 'W', 'X': 'X', 'Y': 'Y', 'Z': 'Z'
  };

  const digitMap = {
    '0': 'nol', '1': 'satu', '2': 'dua', '3': 'tiga',
    '4': 'empat', '5': 'lima', '6': 'enam', '7': 'tujuh',
    '8': 'delapan', '9': 'sembilan'
  };

  let spoken = 'Nomor ';
  spoken += letterMap[letter.toUpperCase()] + ' ';
  digits.forEach(d => {
    spoken += digitMap[d] + ' ';
  });

  const utterance = new SpeechSynthesisUtterance(spoken.trim());
  utterance.lang = 'id-ID';
  utterance.rate = 0.9;

  // Pilih suara Indonesia jika tersedia
  const voices = window.speechSynthesis.getVoices();
  const indoVoice = voices.find(v => v.lang === 'id-ID');
  if (indoVoice) {
    utterance.voice = indoVoice;
  }

  window.speechSynthesis.cancel();
  window.speechSynthesis.speak(utterance);
}

✅ Solusi: Pilih Suara Bahasa Indonesia Secara Eksplisit

Update fungsi speakNumber() agar memaksa pilih suara id-ID dari speechSynthesis.getVoices(), dan menunda bicara sampai suara tersedia.

Ganti seluruh bagian <script> kamu menjadi ini:

html
<script>
  function speakNumber(number) {
    if (!('speechSynthesis' in window)) {
      console.warn('Speech Synthesis API tidak didukung di browser ini.');
      return;
    }

    const letter = number.charAt(0);
    const digits = number.slice(1).split('');

    const letterMap = {
      'A': 'A', 'B': 'B', 'C': 'C', 'D': 'D', 'E': 'E',
      'F': 'F', 'G': 'G', 'H': 'H', 'I': 'I', 'J': 'J',
      'K': 'K', 'L': 'L', 'M': 'M', 'N': 'N', 'O': 'O',
      'P': 'P', 'Q': 'Q', 'R': 'R', 'S': 'S', 'T': 'T',
      'U': 'U', 'V': 'V', 'W': 'W', 'X': 'X', 'Y': 'Y', 'Z': 'Z'
    };

    const digitMap = {
      '0': 'nol', '1': 'satu', '2': 'dua', '3': 'tiga',
      '4': 'empat', '5': 'lima', '6': 'enam', '7': 'tujuh',
      '8': 'delapan', '9': 'sembilan'
    };

    let spoken = 'Nomor ';
    spoken += letterMap[letter.toUpperCase()] + ' ';
    digits.forEach(d => {
      spoken += digitMap[d] + ' ';
    });

    const utterance = new SpeechSynthesisUtterance(spoken.trim());
    utterance.lang = 'id-ID';
    utterance.rate = 0.9;

    // Fungsi pilih suara bahasa Indonesia jika ada
    function setIndoVoiceAndSpeak() {
      const voices = speechSynthesis.getVoices();
      const indoVoice = voices.find(v => v.lang === 'id-ID' || v.name.toLowerCase().includes('indonesia'));

      if (indoVoice) {
        utterance.voice = indoVoice;
      } else {
        console.warn('Suara bahasa Indonesia tidak ditemukan, gunakan default.');
      }

      speechSynthesis.cancel();
      speechSynthesis.speak(utterance);
    }

    // Tunggu suara tersedia jika belum ada
    if (speechSynthesis.getVoices().length === 0) {
      speechSynthesis.onvoiceschanged = setIndoVoiceAndSpeak;
    } else {
      setIndoVoiceAndSpeak();
    }
  }

  window.onload = function () {
    const nomor = document.getElementById('nomor').textContent.trim();
    if (nomor && nomor !== '-') {
      speakNumber(nomor);
    }
  };
</script>

📌 Penjelasan:

  • speechSynthesis.getVoices() butuh waktu untuk siap, maka kita gunakan speechSynthesis.onvoiceschanged.

  • Kita cari suara dengan lang === 'id-ID' atau nama mengandung indonesia → agar fleksibel di berbagai sistem.

  • Jika tidak ada, tetap jalan pakai suara default (meskipun kemungkinan campur bahasa).

<script>
  const digitMap = {
    '0': 'nol', '1': 'satu', '2': 'dua', '3': 'tiga',
    '4': 'empat', '5': 'lima', '6': 'enam', '7': 'tujuh',
    '8': 'delapan', '9': 'sembilan'
  };

  function playAudio(nomor) {
    if (!nomor || nomor === '-') return;

    const letter = nomor.charAt(0).toUpperCase();
    const digits = nomor.slice(1).split('').map(d => digitMap[d]);

    const files = ['nomor', letter, ...digits];

    let index = 0;

    function playNext() {
      if (index >= files.length) return;

      const audio = new Audio(`audio/${files[index]}.mp3`);
      audio.onended = () => {
        index++;
        playNext();
      };
      audio.onerror = () => {
        console.error("Gagal memuat: " + files[index] + ".mp3");
        index++;
        playNext();
      };
      audio.play();
    }

    playNext();
  }

  // Jalankan otomatis saat halaman load
  window.onload = function () {
    const nomor = document.getElementById('nomor').textContent.trim();
    playAudio(nomor);
  };
</script>

</body>
